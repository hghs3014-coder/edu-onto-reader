# 객체/데이터 속성
ex:hasArea a owl:ObjectProperty ;
    rdfs:domain ex:Subject ; rdfs:range ex:LearningArea ; rdfs:label "포함 영역"@ko .
ex:hasCoreIdea a owl:ObjectProperty ;
    rdfs:domain ex:LearningArea ; rdfs:range ex:CoreIdea ; rdfs:label "핵심 아이디어"@ko .
ex:hasContentElement a owl:ObjectProperty ;
    rdfs:domain ex:LearningArea ; rdfs:range ex:ContentElement ; rdfs:label "내용 요소"@ko .
ex:developsCompetency a owl:ObjectProperty ;
    rdfs:domain ex:Subject ; rdfs:range ex:Competency ; rdfs:label "함양 역량"@ko .
ex:relatedToArea a owl:ObjectProperty ;
    rdfs:domain ex:LearningArea ; rdfs:range ex:LearningArea ; rdfs:label "관련 영역"@ko .
ex:gradeBand a owl:DatatypeProperty ;
    rdfs:domain ex:ContentElement ; rdfs:range xsd:string ; rdfs:label "학년군"@ko .
ex:description a owl:DatatypeProperty ;
    rdfs:range xsd:string ; rdfs:label "설명"@ko .
